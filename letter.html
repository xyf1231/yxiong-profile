Letters 手写文字

输入文字，生成手写风格 SVG 路径动画

在主页加载页面使用

把下面这段代码放到 index.html 的加载层里，即可把 Letters 手写动画作为站点加载动效。 已自动适配 site-loading-letters 容器，和 js/script.js 中的逻辑一致。 

<!-- 加载动画容器 -->
<div id="site-loading-letters"
     data-stroke-width="2"></div>

<!-- Letters UMD 产物 -->
<script src="js/letters-dist/letters-animation.umd.js?v=v1.12.2"></script>
<script>
  (function () {
    const el = document.getElementById("site-loading-letters");
    if (!el || typeof window.mountLettersAnimation !== "function") return;

    // 单词列表（已随机打乱）与候选渐变预设
    const words = ["Hello", "Welcome", "Loading", "Coming"];
    const presets = ["neon", "rainbow", "bubblegum"];
    const duration = 2;
    const erase = true;
    let index = 0;
    let timer = null;

    function playNext() {
      if (index >= words.length) {
        index = 0;
      }
      const word = words[index++];
      const preset = presets[Math.floor(Math.random() * presets.length)];
      window.mountLettersAnimation(
        el,
        word,
        preset,
        parseFloat(el.dataset.strokeWidth || "2"),
        0,
        0,
        duration,
        erase
      );
    }

    playNext();
    timer = setInterval(playNext, (erase ? duration * 1.5 : duration) * 1000);
  })();
</script>
